package com.voyah.update;

import android.app.Application;
import android.content.ComponentName;
import android.content.Context;
import android.content.Intent;
import android.os.UserHandle;

public class UpdateApplication extends Application {
    private static final String TAG = "Application";
    public static final String PKGNAME = "com.voyah.update";
    private static final String UPDATE_SERVICE = "com.voyah.update.service.UpdateService";

    private static Context sGlobalContext;

    @Override
    public void onCreate() {
        super.onCreate();
        sGlobalContext = getApplicationContext();
        startUpdateService();
    }

    private void startUpdateService() {
        Intent intent = new Intent();
        intent.setComponent(new ComponentName(PKGNAME, UPDATE_SERVICE));
        sGlobalContext.startServiceAsUser(intent, UserHandle.SYSTEM);
    }
}
